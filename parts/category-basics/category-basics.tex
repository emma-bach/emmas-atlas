\chapter{Categories, Functors and Natural Transformations}
\section{The Language of Categories}
\begin{importantdefinition}{Categories}{}
	A category $\cC$ consists of:
	\begin{enumerate}
		\item A class $\ob \cC$ of \tib{objects},
		\item and a class $\mor \cC$ of \tib{morphisms}, generally consisting of structure-preserving maps between objects,
	\end{enumerate}
	such that:
	\begin{enumerate}
		\item Each morphism $f \in \mor \cC$ has specified objects as its \tib{domain} and \tib{codomain}, with the notation $f : X \to Y$ denoting a morphism with $X$ as its domain and $Y$ as its codomain. The collection of all morphisms $X \to Y$ is called the \tib{hom-set from $X$ to $Y$} (often regardless of whether it ends up being an actual set or a proper class), and is ´generally denoted $\Hom(X,Y)$.
		\item Each object $X \in \ob \cC$ has a designated \tib{identity morphism} $\id_X : X \to X$.
		\item For any pair of morphisms $f,g \in \mor \cC$ such that $\cod f = \dom g$, there exists a \tib{composite morphism} $g \circ f : \dom f \to \cod g$:
		\begin{align*}
			&\forall f : X \to Y, g : Y \to Z :\\ &\exists g \circ f : X \to Z
		\end{align*}
		\item Composition with identity morphisms leaves a morphism unchanged:
		\begin{align*}
			&\forall f : X \to Y:\\ 
			&\id_Y \circ f = f \circ \id_X = f
		\end{align*}
		\item Composition is associative:
		\begin{align*}
			&\forall f : X \to Y, g : Y \to Z, h : Z \to W\\ 
			&(h \circ g) \circ f = h \circ (g \circ f) =: h \circ g \circ f
		\end{align*}
	\end{enumerate}
\end{importantdefinition}
By taking objects as vertices and morphisms as edges, each category can be turned into a graph. This graph is directed, may contain parallel edges, and contains at least one loop at each vertex (given by the identity morphism). We call such a graph a \tib{reflexive quiver}.
\begin{example}
	Many familiar notions of objects equipped with structure-preserving maps define categories:
	\begin{enumerate}
		\item The category $\Set$ has sets as its objects and functions as its morphisms.
		\item The category $\Poset$ as partially ordered sets as its objects and order-preserving functions as morphisms.
		\item The category $\Top$ has topological spaces as its objects and continuous functions as its morphisms.
		\item The categories $\Set_*$ and $\Top_*$ have sets or topological spaces with a designated \tib{base point} as objects and basepoint-preserving (continuous) functions as their morphisms.
		\item The category $\Group$ has groups as its objects and group homomorphisms as its morphisms. Analogously, the categories $\Ring$ and $\Field$ have their respective homomorphism as their morphisms.
		\item For a fixed field $\bk$, $\Vec_\bk$ denotes the category of $\bk$-vector spaces, with linear maps as their morphisms.
		\item For a fixed ring $R$, $_R \Mod$ and $\Mod_R$ denote the categories of left $R$-modules and right $R$-modules, both equipped with $R$-module homomorphisms as their morphisms.
	\end{enumerate}
\end{example}
These sorts of categories, where objects have underlying sets and whose morphism are functions between those sets, are known as \tib{concrete categories} - a precise definition of this concept will follow later. However, non-concrete categories are also prevalent:
\begin{example}
	\begin{enumerate}
		\item For a ring $R$, $\Mat_R$ is the category whose objects are positive integers and where a morphism from $n$ to $m$ is an $m \times n$-matrix with values in $R$.
		\item Any monoid $M$ defines a category $\ttt{B}M$ with a single object, with monoid elements as morphisms and the monoid operation as composition. Vice versa, applying this contruction in reverse allows us to view any single-object category as a monoid.
		\item Any preorder $(P, \leq)$ can be viewed as a category, with elements of $P$ as objects and a unique morphism $x \to y$ if and only if $x \leq y$. In this case, transitivity implies that composite morphisms exist and reflexivity implies that identity morphisms exist.
	\end{enumerate}
\end{example}
\begin{definition}
	We call a category \tib{small} if its class of morphisms forms a set, and \tib{locally small} if the class of morphisms between any pair of objects forms a set.
\end{definition}
\begin{importantdefinition}{}{}
	\begin{enumerate}
		\item An \tib{isomorphism} in a category is a morphism $f : X \to Y$ which has an inverse, i.e. there exists a morphism $g : Y \to X$ such that $g \circ f = \id_X$ and $f \circ g = \id_Y$.
		\item An \tib{endomorphism} in a category is a morphism whose domain equals its codomain.
		\item An endomorphism that is also an isomorphism is called an \tib{automorphism}.
	\end{enumerate}
\end{importantdefinition}
\begin{proposition}
	\phantom{}
	\begin{enumerate}
		\item The Isomorphisms in $\Set$ are precisely the bijections.
		\item The Isomorphisms in $\Group$, $\Ring$, $\Field$ and $_R \Mod$ are the respective algebraic isomorphisms, which end up being exactly the bijective homomorphisms.
		\item The Isomorphisms in $\Top$ are the homeomorphisms - in this case, not all bijective morphisms (i.e. continuous bijections) are isomorphisms.
	\end{enumerate}
\end{proposition}
\begin{exercise}
	In a partial order category $(P, \leq)$, the only isomorphisms are the identity morphisms.
\end{exercise}
\begin{definition}
	A category in which every morphism is an isomorphism is known as a \tib{groupoid}.
\end{definition}
We saw earlier that monoids correspond to single-object categories. If we turn a group into a single-object category using the same construction, we find that the existence of group inverses guarantees that every morphism in this category is an isomorphism. Thus, a groupoid is the multi-object equivalent of a group. Category theorists with a sufficient amount of whimsy in their hearts may argue that categories are similarly multi-object monoids, and could thus be referred to as \tib{monoidoids}.
\begin{proposition}
	Taking a category $\cC$ and ignoring all morphisms that aren't isomorphisms defines a new category, called the \tib{maximal subgroupoid} or \tib{core groupoid} of $\cC$.
\end{proposition}
\chapter{Limits and Colimits}
\begin{importantdefinition}{Cones}{}
	Given some commutative diagram $D$ in a category $\cC$:
	\[\begin{tikzcd}
		X && Y
		\arrow["f", from=1-1, to=1-3]
	\end{tikzcd}\]
	A \tib{cone over $D$} consists of an additional object $c$, alongside a family of morphisms $\Phi$ from $c$ into every object in the diagram, such that the new diagram commutes:
	\[\begin{tikzcd}
		& c & \\
		\\
		X && Y
		\arrow["{\Phi_X}", from=1-2, to=3-1]
		\arrow["{\Phi_Y}"', from=1-2, to=3-3]
		\arrow["f", from=3-1, to=3-3]
	\end{tikzcd}\]
	A \tib{cone under $D$} is similarly defined, but with morphisms going into $c$:
	\[\begin{tikzcd}
		X && Y \\
		\\
		& C
		\arrow["f", from=1-1, to=1-3]
		\arrow["{\Phi_X}"', from=1-1, to=3-2]
		\arrow["{\Phi_Y}", from=1-3, to=3-2]
	\end{tikzcd}\]
\end{importantdefinition}
\begin{importantdefinition}{Limits and Colimits}{}
	A \tib{limit of a commutative diagram $D$} is a \tib{universal cone over $D$}, i.e. a cone $(\lim D, \Phi)$ such that for every other cone $(c, \Psi$, there exists a unique morphism $u : c \to \lim F$ such that $\Phi_X \circ u = \Psi_x$ for all objects $X$ in the diagram:
	\[\begin{tikzcd}
		&& c && \\
		\\
		&& {\lim D} \\
		\\
		X &&&& Y
		\arrow["\exists!u", dotted, from=1-3, to=3-3]
		\arrow["{\Psi_X}"', from=1-3, to=5-1]
		\arrow["{\Psi_Y}", from=1-3, to=5-5]
		\arrow["{\Phi_X}", from=3-3, to=5-1]
		\arrow["{\Phi_Y}"', from=3-3, to=5-5]
		\arrow["f", from=5-1, to=5-5]
	\end{tikzcd}\]
	A \tib{colimit of $D$} is the equivalent construction using a cone under $D$:

	\[\begin{tikzcd}
		X &&&& Y \\
		\\
		&& {\lim D} \\
		\\
		&& c
		\arrow["f", from=1-1, to=1-5]
		\arrow["{\Phi_X}", from=1-1, to=3-3]
		\arrow["{\Psi_X}"', from=1-1, to=5-3]
		\arrow["{\Phi_Y}"', from=1-5, to=3-3]
		\arrow["{\Psi_Y}", from=1-5, to=5-3]
		\arrow["\exists!u"', dotted, from=3-3, to=5-3]
	\end{tikzcd}\]
\end{importantdefinition}

\begin{definition}
	Take a commutative diagram consisting of two objects with two morphisms from one into the other:
	% https://q.uiver.app/#q=WzAsMixbMCwwLCJYIl0sWzIsMCwiWSJdLFswLDEsImYiLDJdLFswLDEsImciLDAseyJvZmZzZXQiOi0zfV1d
	\[\begin{tikzcd}
		X && Y
		\arrow["f"', from=1-1, to=1-3]
		\arrow["g", shift left=3, from=1-1, to=1-3]
	\end{tikzcd}\]
	The limit of this diagram is called its \tib{equalizer} $E$. Since cones have to commute, it suffices to specify only the morphism $i : E \to X$.
	% https://q.uiver.app/#q=WzAsNCxbMiwyLCJYIl0sWzQsMiwiWSJdLFswLDIsIkUiXSxbMCwwLCJGIl0sWzAsMSwiZiIsMl0sWzAsMSwiZyIsMCx7Im9mZnNldCI6LTN9XSxbMiwwLCJpIiwwLHsic3R5bGUiOnsidGFpbCI6eyJuYW1lIjoiaG9vayIsInNpZGUiOiJ0b3AifX19XSxbMywyLCJcXGV4aXN0cyF1IiwwLHsic3R5bGUiOnsiYm9keSI6eyJuYW1lIjoiZG90dGVkIn19fV0sWzMsMCwiaCJdXQ==
	\[\begin{tikzcd}
		F &&&& \\
		\\
		E && X && Y
		\arrow["{\exists! u}", dotted, from=1-1, to=3-1]
		\arrow["h", from=1-1, to=3-3]
		\arrow["i", hook, from=3-1, to=3-3]
		\arrow["f"', from=3-3, to=3-5]
		\arrow["g", shift left=3, from=3-3, to=3-5]
	\end{tikzcd}\]
	The universality of limits now tells us that any morphism $h : F \to X$ such that $f \circ h = g \circ h$ has a unique factorization $u$ through $i$. Note that $i$ has to be a monomorphism, since, given two morphisms $f_1, f_2 : F \to E$, uniqueness of $u$ implies $i \circ f_1 = i \circ f_2 \implies u_1 = u_2 = f_1 = f_2$.
\end{definition}
\begin{example}
	We want to find the equalizer of a group homomorphism $\Phi : G \to H$ and the trivial homomorphism $\epsilon : G \to H$. This means we need
	\begin{align*}
		e_H = \epsilon(i(x)) = \Phi(i(x))
	\end{align*}
	for any $x \in E$. This already tells us that we need $i(E) \subseteq \ker \Phi \subseteq X$. $i : E \to X$ is an injective group homomorphism, and since we can take $F = \ker \Phi$ and $h : \ker \Phi \inj X$ the trivial embedding, $h$ can only factor through $i$ if $i(E) = \ker \Phi$. Thus, $i : E \to \ker \Phi$ is an isomorphism and our equalizer is given by $\ker \Phi$.
\end{example}
\section{Quotient Objects}
\begin{importantdefinition}{Quotient Object}{}
	The quotient object of a morphism $f$ is the coequalizer of the kernel pair of $f$.
\end{importantdefinition}
